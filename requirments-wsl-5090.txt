# insect/script — Windows 主机 + WSL2 + RTX 5090（Blackwell）GPU 推理依赖
#
# 适用：
#   - 宿主机 Windows 10/11，独显 RTX 5090 / 5090 Laptop
#   - 在 **WSL2（Ubuntu 等）** 里跑 Gradio / predict_all（不是本机 PowerShell）
#   - WSL 内 `nvidia-smi` 可见 GPU；驱动 CUDA Version 报告 **13.x**
#
# 与 requirments-windows-5090.txt 的差异：
#   - 本文件面向 **Linux 用户态（WSL）**；torch 轮子是 manylinux，不是 Win 的 .whl
#   - 对 pandas / numpy 强制预编译 wheel，避免「无 gcc → meson 编 pandas」失败
#   - pandas 下限 **2.3.3**（首个带 **cp314** manylinux 轮子的 2.x），适配 WSL 系统 Python 3.14
#
# ⚠ Python：推荐 3.12；本文件也覆盖 **3.14**（torch/numpy/pandas/ORT 均有 cp314 轮子）。
#   勿钉 pandas==2.2.3：该版无 cp314，pip 会报 No matching distribution。
#   勿装「无 wheel 的旧 pandas」再靠 gcc 硬编（meson 报 Unknown compiler）。
#
# ── WSL 内安装（仓库根或 insect/ 下均可）──────────────────────────────────
#   # 强烈建议 venv，避免 --break-system-packages 污染系统包：
#   python3 -m venv .venv && source .venv/bin/activate
#   python -m pip install -U pip
#   pip uninstall -y torch torchvision torchaudio onnxruntime onnxruntime-gpu
#   pip install -r insect/script/requirments-wsl-5090.txt
#
#   # 若坚持用系统 Python 3.14（不推荐）：
#   python3 -m pip install -r insect/script/requirments-wsl-5090.txt --break-system-packages
#
# 若阿里云镜像缺某个 manylinux wheel 又回退到 sdist，可改走官方 PyPI 兜底：
#   pip install -r insect/script/requirments-wsl-5090.txt \
#     --extra-index-url https://pypi.org/simple
#
# ── 验证 ──────────────────────────────────────────────────────────────────
#   python -c "import torch; print(torch.__version__, torch.cuda.is_available(), torch.version.cuda, torch.cuda.get_device_name(0) if torch.cuda.is_available() else '-')"
#   python -c "import onnxruntime as ort; print(ort.get_available_providers())"
#   期望：is_available=True；providers 含 CUDAExecutionProvider
#
# 说明：
#   - WSL2 需启用 GPU 直通；若 nvidia-smi 在 WSL 内失败，先修驱动/WSL。
#   - PyTorch 包名 cu130 = CUDA 13.0 运行时；无需在 WSL 内另装完整 CUDA Toolkit。
#   - 勿与 requirments-windows-5090.txt 混装到同一 venv（Win / Linux 轮子互不兼容）。

--index-url https://mirrors.aliyun.com/pypi/simple/
--trusted-host mirrors.aliyun.com
--extra-index-url https://download.pytorch.org/whl/cu130
--extra-index-url https://pypi.org/simple
# 优先 / 仅用二进制；pandas 无 wheel 时直接失败，避免 meson+gcc 源码编译
--prefer-binary
--only-binary=pandas,numpy

# --- PyTorch（CUDA 13.0 / RTX 5090 · Linux/WSL 轮子；含 cp314）---
torch==2.10.0+cu130
torchvision==0.25.0+cu130

# --- 核心数值 / 图像 ---
# pandas：2.3.3 起有 cp314 manylinux wheel；上限 <3 避开 pandas 3 行为变化
numpy>=1.23.0,<3
pandas>=2.3.3,<3
opencv-python>=4.8.0
Pillow>=10.0.0

# --- YOLO（ultralytics）---
ultralytics==8.4.41

# --- ONNX Runtime GPU（WSL=Linux；1.24+ 有 cp314 轮子）---
# 勿同时安装 onnxruntime（CPU）；二者冲突且会抢包名
onnxruntime-gpu>=1.24.0

# --- Gradio + FastAPI REST ---
gradio>=5.31.0,<6.0.0
fastapi>=0.100.0
uvicorn[standard]>=0.23.0

# --- 验证报表 ---
matplotlib>=3.7.0
